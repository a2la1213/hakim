/* Cockpit v2 — shell commun : esc(), nav, horloge, statut daemon, reduced-motion. */
window.esc = (s) => String(s ?? '').replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
window.REDUCED = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

window.initNav = function (active) {
  document.body.classList.add('shell');
  const views = [['dashboard', '/', 'DASHBOARD'], ['brain', '/brain', 'CERVEAU 3D'], ['dreams', '/dreams', 'LABO DES RÊVES'], ['memory', '/memory', 'MÉMOIRE']];
  const nav = document.createElement('nav');
  nav.className = 'topnav';
  nav.innerHTML = '<div class="beacon"></div><span class="brand">AIOS</span>'
    + views.map(([k, href, label]) => `<a class="pill ${k === active ? 'active' : ''}" href="${href}">${label}</a>`).join('')
    + '<span class="status" id="nav-status">● …</span>';
  document.body.prepend(nav);
  const st = nav.querySelector('#nav-status');
  const tick = () => {
    const h = new Date().toTimeString().slice(0, 8);
    st.textContent = `● ${st.dataset.d || 'daemon ?'} · ${h}`;
  };
  setInterval(tick, 1000); tick();
  const poll = async () => {
    try {
      const a = await (await fetch('/api/anamnese')).json();
      const up = a.daemon_started && (Date.now() / 1000 - a.daemon_started) > 0;
      st.dataset.d = up ? 'daemon up' : 'daemon ?';
      st.classList.toggle('down', !up);
    } catch (_) { st.dataset.d = 'hors ligne'; st.classList.add('down'); }
  };
  poll(); setInterval(poll, 60000);
};

/* Cockpit v2 — overlay de détail universel : la fin du texte tronqué. */
(function () {
  let ov = null;
  function close() { if (ov) { ov.remove(); ov = null; document.removeEventListener('keydown', onKey); } }
  function onKey(e) { if (e.key === 'Escape') close(); }

  window.showDetail = function ({ title, badges = [], meta = [], text = '', voice = false }) {
    close();
    ov = document.createElement('div');
    ov.className = 'overlay';
    ov.innerHTML = `<div class="overlay-panel">
      <span class="close" title="fermer (Échap)">✕</span>
      <h3>${esc(title)}</h3>
      <div class="meta">${badges.map(b => `<span class="badge ${esc(b.cls || '')}">${esc(b.txt)}</span>`).join('')}</div>
      <div class="meta">${meta.filter(([, v]) => v !== null && v !== undefined && v !== '').map(([k, v]) => `<span class="kv">${esc(k)} <b>${esc(v)}</b></span>`).join('')}</div>
      <div class="full-text ${voice ? 'voice' : ''}">${esc(text)}</div>
    </div>`;
    ov.addEventListener('click', (e) => { if (e.target === ov || e.target.classList.contains('close')) close(); });
    document.addEventListener('keydown', onKey);
    document.body.appendChild(ov);
  };

  window.showEpisode = async function (id) {
    id = Number(id);
    if (!Number.isInteger(id) || id < 0) return;
    try {
      const ep = await (await fetch('/api/memory?id=' + id)).json();
      if (ep.error) return;
      const d = ep.ts ? new Date(ep.ts * 1000).toLocaleString('fr-FR') : '';
      window.showDetail({
        title: `Souvenir #${ep.id} · ${ep.role || ''}`,
        badges: (ep.facets || []).slice(0, 8).map(f => ({ txt: f })),
        meta: [['date', d], ['rubrique', ep.rubric], ['importance', ep.importance != null ? (+ep.importance).toFixed(2) : null],
               ['rappels', ep.recall_count], ['valence', ep.enc_valence != null ? (+ep.enc_valence).toFixed(2) : null],
               ['éveil', ep.enc_arousal != null ? (+ep.enc_arousal).toFixed(2) : null]],
        text: ep.content || '',
        voice: ['thought', 'reflection', 'dream', 'reasoning'].includes(ep.role),
      });
    } catch (_) { /* réseau : silencieux */ }
  };
})();

function svgLine(pts, w, h, color, glow) {
  if (!pts.length) return '';
  const xs = (i) => (i / Math.max(1, pts.length - 1)) * (w - 20) + 10;
  const lo = Math.min(...pts), hi = Math.max(...pts), rng = (hi - lo) || 1;
  const ys = (v) => h - 14 - ((v - lo) / rng) * (h - 28);
  const d = pts.map((v, i) => `${i ? 'L' : 'M'}${xs(i).toFixed(1)},${ys(v).toFixed(1)}`).join(' ');
  return `<path d="${d}" fill="none" stroke="${color}" stroke-width="1.8" ${glow ? `style="filter:drop-shadow(0 0 4px ${color})"` : ''}/>`;
}
